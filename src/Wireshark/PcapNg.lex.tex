structure{
  name: Block,
  type_parameters: [],
  parameters: [],
  fields: [
    {name: index, type: Nat},
    {name: kind, type: BlockKind},
    {name: totalLength, type: Nat},
    {name: bodyLength, type: Nat}
  ]
}

definition{
  name: wellFormedBlockLengths,
  parameters: [block : Block],
  result: Bool,
  body: ((12 <=? (project block totalLength)) && (((project block bodyLength) <=? (project block totalLength)) && ((project block bodyLength) <=? 262144)))
}

definition{
  name: isAlignedTo32BitBoundary,
  parameters: [block : Block],
  result: Bool,
  body: ((prim remainder (project block totalLength) 4 0 : Nat) =? 0)
}

definition{
  name: isValidBlock,
  parameters: [block : Block],
  result: Bool,
  body: ((wellFormedBlockLengths block) && (isAlignedTo32BitBoundary block))
}

definition{
  name: alignedEnhancedPacketBlock,
  parameters: [],

}

theorem{
  name: aligned_packet_block_is_valid,
  parameters: [],
  statement: ((isValidBlock (alignedEnhancedPacketBlock)) = true),
  proof: reflexivity{}
}

theorem{
  name: misaligned_packet_block_is_rejected,
  parameters: [],

  proof: reflexivity{}
}
