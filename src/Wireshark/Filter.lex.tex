\title{Wireshark display filter model}
\begin{semanticmodule}
inductive{
  name: ComparisonOp,
  type_parameters: [],
  parameters: [],
  constructors: [
    {name: equal, fields: []},
    {name: greaterThan, fields: []},
    {name: lessThan, fields: []}
  ]
}

inductive{
  name: FilterExpr,
  type_parameters: [],
  parameters: [],
  constructors: [
    {name: literal, fields: [Nat]},
    {name: comparison, fields: [ComparisonOp, Nat, Nat]},
    {name: conjunction, fields: [FilterExpr, FilterExpr]},
    {name: negation, fields: [FilterExpr]}
  ]
}

definition{
  name: evaluateFilter,
  parameters: [expression : FilterExpr],
  result: Bool,
  body: (match expression
    | FilterExpr.literal value => (0 <? value)
    | FilterExpr.comparison op left right => (match op
      | ComparisonOp.equal => (left =? right)
      | ComparisonOp.greaterThan => (right <? left)
      | ComparisonOp.lessThan => (left <? right))
    | FilterExpr.conjunction left right => ((evaluateFilter left) && (evaluateFilter right))
    | FilterExpr.negation inner => (not (evaluateFilter inner))),
  executable: true,
  recursive_argument: expression
}

definition{
  name: filterNodeCount,
  parameters: [expression : FilterExpr],
  result: Nat,
  body: (match expression
    | FilterExpr.literal value => 1
    | FilterExpr.comparison op left right => 1
    | FilterExpr.conjunction left right => (1 + ((filterNodeCount left) + (filterNodeCount right)))
    | FilterExpr.negation inner => (1 + (filterNodeCount inner))),
  executable: true,
  recursive_argument: expression
}

definition{
  name: tcpPortFilter,
  parameters: [],
  result: FilterExpr,
  body: (ctor FilterExpr.conjunction (ctor FilterExpr.comparison (ctor ComparisonOp.equal) 443 443) (ctor FilterExpr.negation (ctor FilterExpr.comparison (ctor ComparisonOp.greaterThan) 443 1024)))
}

theorem{
  name: tcp_port_filter_evaluates_true,
  parameters: [],
  statement: ((evaluateFilter (tcpPortFilter)) = true),
  proof: reflexivity{}
}

theorem{
  name: tcp_port_filter_has_four_nodes,
  parameters: [],
  statement: ((filterNodeCount (tcpPortFilter)) = 4),
  proof: reflexivity{}
}

theorem{
  name: failed_equality_comparison_is_false,
  parameters: [],
  statement: ((evaluateFilter (ctor FilterExpr.comparison (ctor ComparisonOp.equal) 80 443)) = false),
  proof: reflexivity{}
}
\end{semanticmodule}
