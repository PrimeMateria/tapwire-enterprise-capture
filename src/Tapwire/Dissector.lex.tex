\begin{lexlean}{Tapwire.Dissector}

\useglossary{lexlean.std.nat@1.2.0}
\useglossary{tapwire.titles@1.0.0}
\title{Tapwire dissector tree model}
\begin{semanticmodule}
inductive{
  name: ProtocolLayer,
  type_parameters: [],
  parameters: [],
  constructors: [
    {name: ethernet, fields: []},
    {name: ipv4, fields: []},
    {name: tcp, fields: []},
    {name: udp, fields: []},
    {name: tls, fields: []},
    {name: http, fields: []}
  ]
}

structure{
  name: FieldInstance,
  type_parameters: [],
  parameters: [],
  fields: [
    {name: index, type: Nat},
    {name: parentIndex, type: Nat},
    {name: fieldId, type: Nat},
    {name: depth, type: Nat},
    {name: layer, type: ProtocolLayer}
  ]
}

definition{
  name: protocolLayerRank,
  parameters: [layer : ProtocolLayer],
  result: Nat,
  body: (match layer
    | ProtocolLayer.ethernet => 1
    | ProtocolLayer.ipv4 => 2
    | ProtocolLayer.tcp => 3
    | ProtocolLayer.udp => 3
    | ProtocolLayer.tls => 4
    | ProtocolLayer.http => 5)
}

definition{
  name: hasValidParentOrder,
  parameters: [field : FieldInstance],
  result: Bool,
  body: ((project field parentIndex) <? (project field index))
}

definition{
  name: isWithinDepthBudget,
  parameters: [field : FieldInstance],
  result: Bool,
  body: ((project field depth) <=? 32)
}

definition{
  name: isValidFieldInstance,
  parameters: [field : FieldInstance],
  result: Bool,
  body: (((0 <? (project field fieldId)) && (hasValidParentOrder field)) && (isWithinDepthBudget field))
}

definition{
  name: ipSourceField,
  parameters: [],
  result: FieldInstance,
  body: (record FieldInstance {index := 1, parentIndex := 0, fieldId := 11, depth := 1, layer := (ctor ProtocolLayer.ipv4)})
}

definition{
  name: tcpPortField,
  parameters: [],
  result: FieldInstance,
  body: (record FieldInstance {index := 2, parentIndex := 1, fieldId := 23, depth := 2, layer := (ctor ProtocolLayer.tcp)})
}

theorem{
  name: ip_source_field_is_valid,
  parameters: [],
  statement: ((isValidFieldInstance (ipSourceField)) = true),
  proof: reflexivity{}
}

theorem{
  name: tcp_port_field_is_valid,
  parameters: [],
  statement: ((isValidFieldInstance (tcpPortField)) = true),
  proof: reflexivity{}
}

theorem{
  name: sampled_field_ids_are_distinct,
  parameters: [],
  statement: (((project (ipSourceField) fieldId) =? (project (tcpPortField) fieldId)) = false),
  proof: reflexivity{}
}

theorem{
  name: sampled_protocol_layers_are_ordered,
  parameters: [],
  statement: (((protocolLayerRank (project (ipSourceField) layer)) <? (protocolLayerRank (project (tcpPortField) layer))) = true),
  proof: reflexivity{}
}
\end{semanticmodule}
\end{lexlean}
